DROP DATABASE IF EXISTS bd_machomen;
CREATE DATABASE bd_machomen
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;
USE bd_machomen;

-- =============================================================================
-- TABLAS
-- =============================================================================

-- 1. DISTRITOS
CREATE TABLE tb_distrito (
    iddistrito  INT          AUTO_INCREMENT PRIMARY KEY,
    descripcion VARCHAR(50)  NOT NULL
);

-- 2. CATEGORÍAS
CREATE TABLE tb_categoria (
    idcategoria INT          AUTO_INCREMENT PRIMARY KEY,
    descripcion VARCHAR(50)  NOT NULL,
    estado      INT          DEFAULT 1
);

-- 3. USUARIOS
CREATE TABLE tb_usuario (
    idusuario  INT          AUTO_INCREMENT PRIMARY KEY,
    nombres    VARCHAR(100) NOT NULL,
    apellidos  VARCHAR(100) NOT NULL,
    dni        CHAR(8)      NOT NULL UNIQUE,
    correo     VARCHAR(100) NOT NULL UNIQUE,
    clave      VARCHAR(255) NOT NULL,
    direccion  VARCHAR(200) NULL,
    celular    VARCHAR(15)  NULL,
    iddistrito INT          NULL,
    rol        INT          NOT NULL DEFAULT 2,
    estado     INT          DEFAULT 1,
    fecha_reg  DATETIME     DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (iddistrito) REFERENCES tb_distrito(iddistrito)
);

-- 4. PRODUCTOS
CREATE TABLE tb_producto (
    idproducto      INT            AUTO_INCREMENT PRIMARY KEY,
    descripcion     VARCHAR(100)   NOT NULL UNIQUE,
    stock           INT            NOT NULL DEFAULT 0,
    precio          DECIMAL(10,2)  NOT NULL,
    imagen          VARCHAR(255)   NULL,
    talla_json      TEXT           NULL,
    caracteristicas TEXT           NULL,
    idcategoria     INT            NOT NULL,
    estado          INT            DEFAULT 1,
    FOREIGN KEY (idcategoria) REFERENCES tb_categoria(idcategoria)
);

-- 5. BOLETA (cabecera)
CREATE TABLE tb_boleta (
    idboleta           INT           AUTO_INCREMENT PRIMARY KEY,
    fecha              DATETIME      DEFAULT CURRENT_TIMESTAMP,
    idusuario          INT           NOT NULL,
    tipo_pago          INT           NOT NULL DEFAULT 1,
    nro_operacion      VARCHAR(50)   NULL,
    direccion_envio    VARCHAR(200)  NOT NULL,
    iddistrito         INT           NOT NULL,
    importe_productos  DECIMAL(10,2) DEFAULT 0.00,
    importe_envio      DECIMAL(10,2) DEFAULT 0.00,
    importe_neto       DECIMAL(10,2) DEFAULT 0.00,
    importe_igv        DECIMAL(10,2) DEFAULT 0.00,
    importe_total      DECIMAL(10,2) DEFAULT 0.00,
    estado             INT           DEFAULT 1,
    FOREIGN KEY (idusuario)  REFERENCES tb_usuario(idusuario),
    FOREIGN KEY (iddistrito) REFERENCES tb_distrito(iddistrito)
);

ALTER TABLE tb_boleta AUTO_INCREMENT = 1001;

-- 6. BOLETA DETALLE
CREATE TABLE tb_boleta_det (
    idboleta   INT           NOT NULL,
    idproducto INT           NOT NULL,
    cantidad   INT           NOT NULL,
    preciovta  DECIMAL(10,2) NOT NULL,
    importe    DECIMAL(10,2) NOT NULL,
    talla      VARCHAR(10)   NULL,
    PRIMARY KEY (idboleta, idproducto),
    FOREIGN KEY (idboleta)   REFERENCES tb_boleta(idboleta),
    FOREIGN KEY (idproducto) REFERENCES tb_producto(idproducto)
);


-- 7. CUPONES DE DESCUENTO
CREATE TABLE tb_cupon (
    idcupon     INT           AUTO_INCREMENT PRIMARY KEY,
    codigo      VARCHAR(30)   NOT NULL UNIQUE,
    tipo        INT           NOT NULL DEFAULT 1,
    valor       DECIMAL(10,2) NOT NULL,
    minimo      DECIMAL(10,2) DEFAULT 0,
    usos_max    INT           DEFAULT 0,
    usos        INT           DEFAULT 0,
    estado      INT           DEFAULT 1
);
